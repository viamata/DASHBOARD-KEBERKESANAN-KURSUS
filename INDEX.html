Panduan Pemasangan Google Apps Script

Untuk menyambungkan paparan dashboard ini terus dengan Google Sheet anda (ID: 15Dwik-SWciJmSEwAOcI9a1l4gNldvqidAhjCcCs-jPc), anda hanya perlu menyalin kod bahagian belakang (backend) di bawah dan memasukkannya ke dalam Google Apps Script editor anda.

1

Buka Google Sheets

Buka fail Google Sheet yang mempunyai ID di atas. Klik menu Extensions (Sambungan) > Apps Script.

2

Tampal Kod gs

Gantikan semua kod di dalam fail Code.gs dengan kod yang disediakan di sebelah kanan.

3

Cipta Fail HTML

Cipta fail baru (klik butang +), namakan sebagai index. Tampal seluruh kod fail HTML ini ke dalamnya, kemudian Deploy sebagai Web App!

function doGet() {
  return HtmlService.createTemplateFromFile('index')
      .evaluate()
      .setTitle('Dashboard Keberkesanan Kursus')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function getDashboardData() {
  const sheetId = '15Dwik-SWciJmSEwAOcI9a1l4gNldvqidAhjCcCs-jPc';
  const sheetName = 'Sheet1';
  
  try {
    const sheet = SpreadsheetApp.openById(sheetId).getSheetByName(sheetName);
    if (!sheet) {
      throw new Error("Tab '" + sheetName + "' tidak ditemui.");
    }
    
    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) {
      return {
        success: false,
        message: "Tiada data ditemui di dalam sheet."
      };
    }
    
    const headers = data[0].map(h => h.toString().trim().toUpperCase());
    
    // Cari indeks lajur header mengikut keperluan struktur
    const idxNama = headers.indexOf('NAMA PENUH');
    const idxTempat = headers.indexOf('TEMPAT BERTUGAS');
    const idxSangatBerkesan = headers.indexOf('NILAIKAN KEBERKESANAN KURSUS [SANGAT BERKESAN]');
    const idxBerkesan = headers.indexOf('NILAIKAN KEBERKESANAN KURSUS [BERKESAN]');
    const idxKurangBerkesan = headers.indexOf('NILAIKAN KEBERKESANAN KURSUS [KURANG BERKESAN]');
    const idxTidakBerkesan = headers.indexOf('NILAIKAN KEBERKESANAN KURSUS [TIDAK BERKESAN]');
    const idxSyor = headers.indexOf('ADAKAH ANDA AKAN MENGESYORKAN PADA RAKAN-RAKAN');
    
    let totalPenyertaan = 0;
    const tempatBertugasCount = {};
    let countSangatBerkesan = 0;
    let countBerkesan = 0;
    let countKurangBerkesan = 0;
    let countTidakBerkesan = 0;
    let totalSyorYa = 0;
    let totalSyorRespondents = 0;
    const rawRows = [];

    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      // Abaikan baris kosong yang tiada nama penuh
      if (idxNama !== -1 && !row[idxNama]) continue;
      
      totalPenyertaan++;
      
      // Tempat Bertugas (LOKASI)
      if (idxTempat !== -1) {
        const tempat = row[idxTempat] ? row[idxTempat].toString().trim() : 'Lain-lain';
        if (tempat) {
          tempatBertugasCount[tempat] = (tempatBertugasCount[tempat] || 0) + 1;
        }
      }
      
      // Keberkesanan Kursus
      // Pemeriksaan penanda (Jika ada nilai tak kosong atau bukan 'false' atau bukan '0' atau 'tidak')
      const isMarked = (val) => {
        if (!val) return false;
        const s = val.toString().trim().toLowerCase();
        return s !== '' && s !== 'false' && s !== '0' && s !== 'tidak';
      };

      if (idxSangatBerkesan !== -1 && isMarked(row[idxSangatBerkesan])) countSangatBerkesan++;
      if (idxBerkesan !== -1 && isMarked(row[idxBerkesan])) countBerkesan++;
      if (idxKurangBerkesan !== -1 && isMarked(row[idxKurangBerkesan])) countKurangBerkesan++;
      if (idxTidakBerkesan !== -1 && isMarked(row[idxTidakBerkesan])) countTidakBerkesan++;
      
      // Syor Rakan
      if (idxSyor !== -1) {
        const syorVal = row[idxSyor] ? row[idxSyor].toString().trim().toLowerCase() : '';
        if (syorVal !== '') {
          totalSyorRespondents++;
          if (syorVal === 'ya' || syorVal === 'yes' || syorVal === 'true' || syorVal === '1' || syorVal === 'mengesyorkan') {
            totalSyorYa++;
          }
        }
      }

      // Simpan beberapa rekod raw data untuk visualisasi senarai responden
      rawRows.push({
        nama: idxNama !== -1 ? row[idxNama] : '',
        tempat: idxTempat !== -1 ? row[idxTempat] : '',
        syor: idxSyor !== -1 ? row[idxSyor] : ''
      });
    }
    
    const peratusSyor = totalSyorRespondents > 0 ? Math.round((totalSyorYa / totalSyorRespondents) * 100) : 0;
    
    return {
      success: true,
      data: {
        totalPenyertaan: totalPenyertaan,
        uniqueTempatBertugas: Object.keys(tempatBertugasCount).length,
        tempatBertugasDist: tempatBertugasCount,
        keberkesanan: {
          sangatBerkesan: countSangatBerkesan,
          berkesan: countBerkesan,
          kurangBerkesan: countKurangBerkesan,
          tidakBerkesan: countTidakBerkesan
        },
        peratusSyor: peratusSyor,
        totalSyorYa: totalSyorYa,
        totalSyorRespondents: totalSyorRespondents,
        rawRows: rawRows.slice(0, 50) // Ambil had 50 responden teratas untuk senarai
      }
    };
    
  } catch (error) {
    return {
      success: false,
      error: error.message
    };
  }
}
